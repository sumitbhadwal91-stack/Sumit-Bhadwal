पहली बार Setup कैसे होगा?
1) Firebase में नया project बनाइए। 2) Web App जोड़कर उसका config firebase-config.js में डालिए। 3) Authentication → Email/Password ON कीजिए। 4) Firestore Database बनाइए। 5) नीचे दिए Security Rules लगाइए। फिर इसी email/password से मोबाइल और computer दोनों में login करें।
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId}/jct/{docId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
